## 0) 数据集（make_inputs.py，seed=45）
$ python3 make_inputs.py
chr1_len=2000000
genes=40
exons=215
cpg=30          # 12 孤立 + 6 簇 x 3 个相互重叠
peaks=10        # narrowPeak 式 BED6+4
transcripts_bed12=8
variants=30
sam_reads=6     # 含 1 条 spliced（50M100N50M）

## 1) genome.txt 从同一参考 FASTA 生成
$ samtools faidx chr1.fa
$ cut -f1,2 chr1.fa.fai > genome.txt
chr1	2000000

## 2) 校验：字段数 / 负或倒置区间 / 染色体名
$ awk -F'\t' '{print NF}' genes.bed | sort -u      # 6（exons/cpg 同为 6）
$ awk -F'\t' '{print NF}' peaks.bed | sort -u      # 10（narrowPeak）
$ awk -F'\t' '{print NF}' transcripts.bed12 | sort -u  # 12
$ awk -F'\t' '$2 < 0 || $2 >= $3' *.bed            # 无输出（无倒置）
$ cut -f1 genes.bed | sort -u                      # chr1

## 3) sort：bedtools sort 与 coreutils 等价
$ bedtools sort -i cpg.bed > cpg.sorted.bed
$ sort -k1,1 -k2,2n cpg.bed > cpg.coreutils.bed
$ diff cpg.sorted.bed cpg.coreutils.bed            # 空 = IDENTICAL

## 4) slop 依赖 genome.txt，边界自动裁剪
$ printf 'chr1\t0\t100\tedge_left\t500\t.\nchr1\t1999500\t2000000\tedge_right\t500\t.\n' > edge.bed
$ bedtools slop -i edge.bed -g genome.txt -b 500
chr1	0	600	edge_left	500	.          # start 顶到 0，不越界
chr1	1999000	2000000	edge_right	500	.      # end 钉在 2000000，不越界

## 5) flank：上游 1 kb 启动子区与 CpG 重叠
$ bedtools flank -i genes.sorted.bed -g genome.txt -l 1000 -r 0 > promoters.bed   # 40 条
$ awk -F'\t' '$7 >= 1' promoters_cpg.txt | wc -l   # 0（本次模拟数据中启动子区无 CpG，真实结果）

## 6) merge：只折叠重叠/首尾相接区间
$ bedtools merge -i cpg.sorted.bed > cpg.merged.bed            # 30 -> 20
$ bedtools merge -i cpg.sorted.bed -d 200 > cpg.merged.d200.bed # 30 -> 19
$ bedtools merge -i genes.sorted.bed                            # 40 -> 40（无重叠，不变）
$ bedtools merge -i exons.sorted.bed                            # 215 -> 215（不变）
$ awk -F'\t' '{s+=$3-$2} END{print s}' cpg.merged.bed           # 30862
$ awk -F'\t' '{s+=$3-$2} END{print s}' cpg.bed                  # 36369（原始含簇内重叠，重复计入 5507 bp）

## 7) intersect：外显子 vs CpG 岛
$ bedtools intersect -a exons.sorted.bed -b cpg.sorted.bed -c > exons_cpg.txt
$ awk -F'\t' '$7>=1' exons_cpg.txt | wc -l          # 7/215 个外显子有 CpG 重叠
$ bedtools intersect -a exons.sorted.bed -b cpg.sorted.bed | wc -l   # 9 对
$ bedtools intersect -a exons.sorted.bed -b cpg.sorted.bed | awk '{s+=$3-$2} END{print s}'  # 994 bp

## 8) complement 覆盖恒等式（intervals + complement = 染色体长）
$ bedtools complement -i cpg.sorted.bed -g genome.txt > non_cpg.bed    # 21 条
30862 (merged CpG) + 1969138 (non-CpG) = 2000000   # 精确成立
365304 (gene bodies) + 1634696 (non-gene) = 2000000 # 精确成立

## 9) shuffle（同样依赖 genome.txt）
$ bedtools shuffle -i cpg.sorted.bed -g genome.txt -seed 45 > cpg.shuffled.bed  # 30 条
$ bedtools intersect -a cpg.sorted.bed -b cpg.shuffled.bed -u | wc -l           # 3（随机落回原位）

## 10) makewindows
$ bedtools makewindows -g genome.txt -w 10000 | wc -l                # 200（固定 10 kb）
$ bedtools makewindows -g genome.txt -w 10000 -s 5000 -i winnum | wc -l  # 400（滑动）
$ tail -1 windows_slide.bed                                          # chr1 1995000 2000000 400

## 11) getfasta：0-based 半开的长度即 end - start
$ bedtools getfasta -fi chr1.fa -bed exons.sorted.bed -name -fo exons.fa
getfasta_records=215 len_eq_end_minus_start=215   # 215/215 全部成立
first=gene001.E1::chr1:30450-30739 len=289        # 30739-30450=289，无 +1

## 12) VCF(1-based) -> BED(0-based)：start-1，end 不变；1 bp 地标往返
$ grep -v '^#' variants.vcf | awk 'BEGIN{OFS="\t"} {print $1, $2-1, $2}' > variants.bed  # 30 条
$ awk 'BEGIN{OFS="\t"} {print $1, $4-1, $4}' landmark.gff     # GFF 1000-1000 -> BED chr1 999 1000
$ samtools faidx chr1.fa chr1:1000-1000   -> T
$ bedtools getfasta -fi chr1.fa -bed landmark.bed -> T
$ VCF POS=1000 REF                                        -> T
三途径碱基一致（T）：1 bp 特征往返测试通过，end 在两种约定下数值相同

## 13) bamtobed：SAM 1-based POS -> BED 0-based start（POS-1）
$ samtools view -bS reads.sam > reads.bam
$ bedtools bamtobed -i reads.bam > alignments.bed
read1: SAM POS=50001 -> BED start=50000      # 6/6 条 read 均满足 start = POS-1
$ bedtools bamtobed -i reads.bam -split > spliced.bed
read2_spliced (CIGAR 50M100N50M) -> 2 块: 150000-150050 / 150150-150200

## 14) BED12 块不变量 + bed12tobed6
$ bedtools bed12tobed6 -i transcripts.bed12 > transcripts_exons.bed6   # 47 条
bed12_models=8 total_blocks=47 invariants_all_pass=True
# 校验项：blockStarts[0]=0；末块 blockStart+blockSize == chromEnd-chromStart；
#         块升序不重叠；thickStart/thickEnd 落在 [chromStart, chromEnd]

## 15) narrowPeak：summit = chromStart + peak（第 10 列，0-based 偏移）
$ awk '{printf "%s\t%d\t%d\t%s\t%d\n",$1,$2,$3,$4,$2+$10}' peaks.bed > summits.tsv
peak01: 575414 + 383 = 575797
peak10: peak 列 = -1（未赋值），start-1 = 1534133 不是真实 summit
# 教训（首跑实测）：误用第 8 列 pValue 求 summit 得到 575429 这类假值；
# narrowPeak 列语义纯靠位置记忆，BED6+4 无表头

## 16) 失败演示 A：染色体名不匹配（chr1 vs 1）——静默空结果

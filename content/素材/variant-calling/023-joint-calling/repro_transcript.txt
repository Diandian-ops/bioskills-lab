# 0. 输入概览
bcftools view -H $IN | wc -l                      # input_records
bcftools query -l $IN | wc -l                     # input_samples
sed -n '1p;420p;840p;1260p;1680p;2100p' all_samples.txt | paste -sd, -
#   → SAMPLES=HG00096,HG01280,HG02220,HG03295,NA11894,NA19320

# 1) 逐样本 callset（两步法：先选样本，再排除该样本为纯合参考的位点；
#    模拟「单样本调用只报变异位点」的产物形态）
for s in $SAMPLES; do
  bcftools view -s $s $IN -Ou | bcftools view -e 'GT="ref"' -Oz -o per_${s}.vcf.gz
  bcftools index -t per_${s}.vcf.gz
done

# 2) 朴素合并（反模式）
bcftools merge -Oz -o merged_naive.vcf.gz per_*.vcf.gz

# 3) 真·联合口径对照（6 样本子集，平方化矩阵）
bcftools view -s $SAMPLES $IN -Oz -o joint6.vcf.gz

# 4) 逐样本 ./. 统计（bcftools query -f '[%GT]\n' + awk 计数）

# 5) GT 一致性：仅在该样本被调用的位点上逐位点比对 merge vs joint

# 6) 汇总 joint_calling_stats.json（python3，供 make_figs.py）

## 7) BAM -> SAM -> BAM 往返
$ samtools view -h -o demo.sam aligned_e2e.bam && samtools view -c demo.sam
6000
$ samtools view -b -o demo_back.bam demo.sam && samtools view -c demo_back.bam
6000

## 8) BAM -> CRAM -> BAM 往返（需同一参考基因组）
$ samtools view -C -T reference.fa -o demo.cram aligned_e2e.bam && samtools view -c demo.cram
6000
$ samtools view -b -T reference.fa -o demo_back2.bam demo.cram && samtools view -c demo_back2.bam
6000
